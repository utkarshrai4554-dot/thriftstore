rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    // Products collection rules
    match /products/{productId} {
      // Anyone can read approved products
      allow read: if resource.data.status == 'approved';
      
      // Only authenticated users can create products
      allow create: if request.auth != null;
      
      // Only seller can update their own products if pending
      // Admin can update any product
      allow update: if (request.auth != null && 
        (resource.data.sellerId == request.auth.uid && resource.data.status == 'pending') ||
        request.auth.token.admin == true);
      
      // Only seller can delete their own products if pending
      // Admin can delete any product
      allow delete: if (request.auth != null && 
        (resource.data.sellerId == request.auth.uid && resource.data.status == 'pending') ||
        request.auth.token.admin == true);
    }
    
    // Likes collection rules
    match /products/{productId}/likes/{likeId} {
      allow read, write: if request.auth != null;
    }
    
    // Users collection rules
    match /users/{userId} {
      allow read, write: if request.auth != null && request.auth.uid == userId;
    }
    
    // Reviews collection rules
    match /reviews/{reviewId} {
      allow read: if true;
      allow write: if request.auth != null;
    }
    
    // Orders collection rules
    match /orders/{orderId} {
      allow read, write: if request.auth != null;
    }
  }
}
